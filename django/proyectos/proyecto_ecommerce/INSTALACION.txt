"""
=============================================================================
              ECOMMERCE DJANGO - INSTALACIÓN PASO A PASO
=============================================================================

Este documento guía la instalación completa del ecommerce desde cero.
Tiempo estimado: 10-15 minutos
"""

# PASO 1: PREPARAR EL ENTORNO
# ============================

# 1a. Abre tu terminal/PowerShell
# 1b. Navega a la carpeta del proyecto:
cd d:\ecommerce\proyecto_ecommerce

# 1c. Crear entorno virtual (si aún no existe):
python -m venv env

# 1d. Activar entorno virtual:
# En Windows:
env\Scripts\activate
# En Linux/macOS:
source env/bin/activate

# PASO 2: INSTALAR DEPENDENCIAS
# ==============================

pip install -r requirements.txt

# O instalar manualmente:
pip install Django==5.2.17 Pillow==10.0.0 python-decouple==3.8

# PASO 3: PREPARAR LA BASE DE DATOS
# ==================================

# 3a. Aplicar migraciones (crear tablas):
python manage.py migrate

# Deberías ver algo como:
# Operations to perform:
#   Apply all migrations: admin, auth, contenttypes, sessions, productos, usuarios, carrito, pedidos
# Running migrations:
#   Applying contenttypes.0001_initial... OK
#   ...y más migraciones

# 3b. Crear superusuario (administrador):
python manage.py createsuperuser

# Te pedirá:
#   Username: admin
#   Email: admin@ejemplo.com
#   Password: (escribe una contraseña segura, mínimo 8 caracteres)
#   Password (again): (repite la contraseña)

# PASO 4: CARGAR DATOS DE PRUEBA (OPCIONAL)
# ===========================================

python manage.py shell < cargar_datos.py

# Esto creará:
# - 3 categorías (Electrónica, Ropa, Libros)
# - 6 productos de ejemplo
# - 2 usuarios de prueba (juan, maria)
# - Direcciones de envío

# PASO 5: INICIAR EL SERVIDOR
# ============================

python manage.py runserver

# Deberías ver:
# Django version 5.2.17, using settings 'config.settings'
# Starting development server at http://127.0.0.1:8000/
# Quit the server with CONTROL-BREAK

# PASO 6: ACCEDER A LA APLICACIÓN
# ================================

# Abre tu navegador y accede a:

# 📍 Sitio principal:
http://localhost:8000/

# 🔑 Panel de administración:
http://localhost:8000/admin/
# Usa el usuario y contraseña que creaste en PASO 3

# 🛍️ Productos:
http://localhost:8000/productos/

# 🛒 Carrito:
http://localhost:8000/carrito/

# PASO 7: EXPLORAR LA APLICACIÓN
# ===============================

# En el panel de admin (/admin/):
# 1. Crear más productos y categorías
# 2. Cambiar estado de pedidos
# 3. Ver usuarios y perfiles
# 4. Explorar las opciones de filtrado y búsqueda

# PASO 8: PRÓXIMOS PASOS
# =====================

# Para seguir desarrollando:

# a) Crear plantillas HTML personalizadas en templates/
# b) Agregar estilos CSS en static/css/
# c) Crear más funcionalidades
# d) Implementar pasarela de pagos (Stripe, PayPal)
# e) Agregar API REST con Django REST Framework
# f) Configurar emails

# COMANDOS ÚTILES
# ===============

# Ver todos los comandos disponibles:
python manage.py help

# Crear nuevas migraciones después de cambiar modelos:
python manage.py makemigrations

# Aplicar las migraciones:
python manage.py migrate

# Ejecutar tests:
python manage.py test

# Ver la base de datos en formato JSON:
python manage.py dumpdata > datos.json

# Restaurar datos desde JSON:
python manage.py loaddata datos.json

# Limpiar caché y archivos temporales:
python manage.py clean_pyc
python manage.py clear_cache

# DETENER EL SERVIDOR
# ====================

# Presiona: CTRL + C en la terminal

# DESACTIVAR ENTORNO VIRTUAL
# ===========================

deactivate
